#!/bin/bash
# =============================================================================
# disable_startup.command  —  MeePo Tek
# Tat cac chuong trinh TechTool (Micromat) tu chay khi boot tren cac o macOS
# gan ngoai (USB). Khong xoa gi: file .plist duoc doi vao
#   <volume>/Users/Shared/disabled-startup/  (giu nguyen duong dan goc ben trong)
# de co the bat lai bang --restore.
#
# Bam dup -> hien menu chon.  Hoac chay lenh:
#   sudo bash disable_startup.command                 # quet TAT CA o ngoai dang gan
#   sudo bash disable_startup.command MeePoCata       # chi 1 (hoac vai) volume
#   sudo bash disable_startup.command -g              # tat luon Google Updater
#   sudo bash disable_startup.command -r MeePoCata    # bat lai (restore)
#
# Viet cho bash 3.2 cua macOS (khong dung tinh nang bash 4+).
# =============================================================================
set -u
shopt -s nullglob

usage() { sed -n '2,16p' "$0" | sed 's/^# \{0,1\}//'; }

# Giu cua so Terminal mo khi chay bang bam dup (de doc ket qua)
pause_end() { if [ -t 0 ]; then echo; read -r -p "Bam Enter de dong..." _; fi; }

# --- Tu xin quyen root (giu nguyen tham so) ---
if [ "$(id -u)" -ne 0 ]; then
  exec sudo bash "$0" "$@"
fi

MODE="disable"
GOOGLE=0
VOLS=()
for a in "$@"; do
  case "$a" in
    -g|--google)  GOOGLE=1 ;;
    -r|--restore) MODE="restore" ;;
    -h|--help)    usage; exit 0 ;;
    -*)           echo "Tham so khong hop le: $a"; usage; exit 1 ;;
    *)            VOLS+=("$a") ;;
  esac
done

# --- Bam dup (khong tham so) -> menu ---
if [ $# -eq 0 ] && [ -t 0 ]; then
  echo "=== TAT TU CHAY KHI BOOT (o macOS gan ngoai) ==="
  echo "  1) Tat TechTool (mac dinh)"
  echo "  2) Tat TechTool + Google Updater"
  echo "  3) Bat lai (restore) nhung muc da tat"
  read -r -p "Chon 1/2/3 [1]: " c
  case "${c:-1}" in
    2) GOOGLE=1 ;;
    3) MODE="restore" ;;
  esac
fi

# --- Tu ten volume -> thu muc goc chua Library/Users ---
# Catalina tro len: "<ten> - Data" (volume ghi duoc); ban cu: chinh volume
resolve_root() {
  local v="${1#/Volumes/}"
  v="${v%/}"
  v="${v% - Data}"
  if [ -d "/Volumes/$v - Data/Library" ]; then
    echo "/Volumes/$v - Data"
  elif [ -d "/Volumes/$v/Library" ]; then
    echo "/Volumes/$v"
  fi
}

# --- Khong co ten volume -> tu tim moi o macOS gan ngoai ---
if [ ${#VOLS[@]} -eq 0 ]; then
  for p in /Volumes/*; do
    [ -L "$p" ] && continue                  # bo o dang chay may (Macintosh HD -> /)
    [ -d "$p" ] || continue
    case "$(basename "$p")" in
      Preboot|Recovery|VM|Update|EFI|"Install macOS"*|Install*) continue ;;
    esac
    VOLS+=("$(basename "$p")")
  done
fi

# --- Loc ra danh sach root, bo trung ---
ROOTS=()
for v in ${VOLS[@]+"${VOLS[@]}"}; do
  r="$(resolve_root "$v")"
  [ -z "$r" ] && continue
  [ -d "$r/Library/LaunchDaemons" ] || [ -d "$r/Library/LaunchAgents" ] || continue
  dup=0
  for x in ${ROOTS[@]+"${ROOTS[@]}"}; do [ "$x" = "$r" ] && dup=1; done
  [ $dup -eq 0 ] && ROOTS+=("$r")
done

if [ ${#ROOTS[@]} -eq 0 ]; then
  echo "Khong tim thay volume macOS nao phu hop (o ngoai co Library/LaunchDaemons)."
  echo "Kiem tra:  ls /Volumes"
  pause_end; exit 1
fi

FAIL_PERM=0
move_file() {  # move_file <nguon> <dich>
  local err
  if err=$(mv "$1" "$2" 2>&1); then return 0; fi
  echo "  X Loi: $err"
  case "$err" in *"Operation not permitted"*) FAIL_PERM=1 ;; esac
  return 1
}

disable_root() {
  local root="$1" dest="$1/Users/Shared/disabled-startup" n=0 f rel
  if ! mkdir -p "$dest" 2>/dev/null; then
    echo "  X Khong tao duoc $dest (volume chi doc?)"; return
  fi
  local items=()
  items+=( "$root"/Library/LaunchDaemons/com.micromat.*.plist )
  items+=( "$root"/Library/LaunchAgents/com.micromat.*.plist )
  items+=( "$root"/Library/LaunchAgents/TechTool*.plist )
  if [ $GOOGLE -eq 1 ]; then
    items+=( "$root"/Library/LaunchAgents/com.google.*.plist )
    items+=( "$root"/Library/LaunchDaemons/com.google.*.plist )
    items+=( "$root"/Users/*/Library/LaunchAgents/com.google.*.plist )
  fi
  for f in ${items[@]+"${items[@]}"}; do
    [ -e "$f" ] || continue
    rel="${f#$root/}"                       # vd: Library/LaunchDaemons/x.plist
    mkdir -p "$dest/$(dirname "$rel")"
    if [ -e "$dest/$rel" ]; then
      echo "  ! Da co ban trong disabled-startup, bo qua: $rel"
      continue
    fi
    move_file "$f" "$dest/$rel" && { echo "  - Da tat: $rel"; n=$((n+1)); }
  done
  echo "  => Tat $n muc."
}

restore_root() {
  local root="$1" dest="$1/Users/Shared/disabled-startup" n=0 rel target
  if [ ! -d "$dest" ]; then
    echo "  Khong co thu muc disabled-startup."
    return
  fi
  while IFS= read -r rel; do
    rel="${rel#./}"
    case "$rel" in
      */*)               target="$root/$rel" ;;
      TechTool*.plist)   target="$root/Library/LaunchAgents/$rel" ;;   # kieu cu (de phang)
      com.micromat.*)    target="$root/Library/LaunchDaemons/$rel" ;;
      *) echo "  ? Khong biet cho goc, bo qua: $rel"; continue ;;
    esac
    if [ -e "$target" ]; then
      echo "  ! Da ton tai, bo qua: ${target#$root/}"
      continue
    fi
    mkdir -p "$(dirname "$target")"
    move_file "$dest/$rel" "$target" && { echo "  + Da bat lai: ${target#$root/}"; n=$((n+1)); }
  done < <(cd "$dest" && find . -type f -name '*.plist')
  echo "  => Bat lai $n muc."
}

for r in "${ROOTS[@]}"; do
  echo "== $r =="
  if [ "$MODE" = "restore" ]; then
    restore_root "$r"
  else
    disable_root "$r"
  fi
done

if [ $FAIL_PERM -eq 1 ]; then
  echo
  echo "! Bi macOS chan (Operation not permitted): cap quyen cho Terminal:"
  echo "  System Settings -> Privacy & Security -> Full Disk Access -> bat Terminal,"
  echo "  dong Terminal roi chay lai."
fi
echo "Xong."
pause_end
